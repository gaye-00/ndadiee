# Sahel UI

A React component library designed for **slow networks and low-end phones**: small bundles, no layout shift, and full keyboard accessibility.

```tsx
import { Button, Dialog, Skeleton } from 'sahel-ui';
import 'sahel-ui/styles.css';
```

| Component | Gzipped |

| Button | 0.6 kB |
| Dialog | 1.9 kB |
| Skeleton | 0.4 kB |

Used in production by three products. The API is stable; the project is in **maintenance mode** and not currently looking for new contributors. Security fixes are still released.

==> src/button/Button.tsx <==
import { forwardRef, type ButtonHTMLAttributes } from 'react';

export interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'ghost';
  /** Shows a spinner and blocks double submissions on slow networks */
  loading?: boolean;
}

export const Button = forwardRef<HTMLButtonElement, ButtonProps>(function Button(
  { variant = 'primary', loading = false, disabled, children, className, ...props },
  ref,
) {
  return (
    <button
      ref={ref}
      // aria-disabled rather than disabled: the button keeps its focus while the request runs
      aria-disabled={disabled || loading || undefined}
      aria-busy={loading || undefined}
      className={['sahel-button', `sahel-button--${variant}`, className].filter(Boolean).join(' ')}
      {...props}
      onClick={disabled || loading ? (event) => event.preventDefault() : props.onClick}
    >
      {loading && <span className="sahel-button__spinner" aria-hidden />}
      {children}
    </button>
  );
});

==> src/skeleton/Skeleton.tsx <==
import type { CSSProperties } from 'react';

interface SkeletonProps {
  /** Reserve the exact size of what is coming: no layout shift when it arrives */
  width: CSSProperties['width'];
  height: CSSProperties['height'];
  radius?: CSSProperties['borderRadius'];
}

export function Skeleton({ width, height, radius = '0.5rem' }: SkeletonProps) {
  return <span className="sahel-skeleton" style={{ width, height, borderRadius: radius }} aria-hidden />;
}

==> src/dialog/Dialog.tsx <==
import { useEffect, useRef, type ReactNode } from 'react';

interface DialogProps {
  open: boolean;
  onClose: () => void;
  title: string;
  children: ReactNode;
}

/** The native <dialog> element: focus trap, Escape and backdrop for free, 1.9 kB. */
export function Dialog({ open, onClose, title, children }: DialogProps) {
  const dialog = useRef<HTMLDialogElement>(null);

  useEffect(() => {
    const element = dialog.current;
    if (!element) return;
    if (open && !element.open) element.showModal();
    if (!open && element.open) element.close();
  }, [open]);

  return (
    <dialog ref={dialog} className="sahel-dialog" aria-labelledby="sahel-dialog-title" onClose={onClose}>
      <h2 id="sahel-dialog-title">{title}</h2>
      {children}
    </dialog>
  );
}
