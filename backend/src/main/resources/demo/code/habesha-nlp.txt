# Habesha NLP

Open tools for **natural language processing in Amharic and Tigrinya**: tokeniser, normaliser for Ge'ez script variants, and a small labelled corpus.

```python
from habesha_nlp import normalize, tokenize

tokenize(normalize("ሰላም ነው፧ እንዴት ነህ።"))
# ['ሰላም', 'ነው', '?', 'እንዴት', 'ነህ', '።']
```

## Why a normaliser?

Ge'ez script has several characters for the same sound (ሀ, ሐ, ኀ all read *ha*). Writers mix them freely, which splits the vocabulary of any model in two or three. `normalize` maps each family to one character.

## Install

```bash
pip install habesha-nlp
```

## Corpus

`corpus/amharic-news.tsv`: 3 000 sentences from public news sources, labelled with their topic. Native speakers who can review annotations are especially welcome (#14).

==> habesha_nlp/normalize.py <==
"""Map Ge'ez homophones to a single character, so that one word has one spelling."""

# Each family of characters that read the same, mapped to its most common member (first order shown;
# the other orders follow the same offsets in the Unicode block)
_FAMILIES = {
    "ሐ": "ሀ",  # ḥa → ha
    "ኀ": "ሀ",  # ḫa → ha
    "ሠ": "ሰ",  # śa → sa
    "ዐ": "አ",  # ʿa → ʾa
    "ፀ": "ጸ",  # ṣ́a → ṣa
}
_ORDERS = 7

_TABLE = {}
for source, target in _FAMILIES.items():
    for order in range(_ORDERS):
        _TABLE[ord(source) + order] = chr(ord(target) + order)

# Ethiopic question mark (፧) is rare outside formal texts: most writers use « ? »
_TABLE[ord("፧")] = "?"


def normalize(text: str) -> str:
    """The same text, with one character per sound."""
    return text.translate(_TABLE)

==> habesha_nlp/tokenize.py <==
"""Split Amharic and Tigrinya text into words and punctuation."""

import re

# Ethiopic word space (፡) and full stop (።), comma (፣), semicolon (፤), colon (፥), plus Latin punctuation
_PUNCTUATION = "።፣፤፥፦፧?!,.;:"
_TOKEN = re.compile(rf"[^\s፡{re.escape(_PUNCTUATION)}]+|[{re.escape(_PUNCTUATION)}]")


def tokenize(text: str) -> list[str]:
    """Words and punctuation marks, in order. The traditional word separator (፡) is treated as a space."""
    return _TOKEN.findall(text)

==> tests/test_normalize.py <==
from habesha_nlp import normalize, tokenize


def test_homophones_share_one_spelling():
    assert normalize("ሐበሻ") == normalize("ሀበሻ")


def test_all_orders_follow_their_family():
    assert normalize("ሑ") == "ሁ"


def test_traditional_separator_splits_words():
    assert tokenize("ሰላም፡ነው።") == ["ሰላም", "ነው", "።"]

==> scripts/dedupe.py <==
"""Drop near-duplicate sentences: news agencies republish the same story with small edits (#19)."""

import csv
import sys

from habesha_nlp import normalize


def key(text: str) -> str:
    # Same sentence once normalised and stripped of punctuation and spaces
    return "".join(ch for ch in normalize(text) if ch.isalnum())


def main(path: str) -> None:
    seen: set[str] = set()
    with open(path, newline="", encoding="utf-8") as source:
        rows = list(csv.DictReader(source, delimiter="\t"))
    writer = csv.DictWriter(sys.stdout, fieldnames=rows[0].keys(), delimiter="\t")
    writer.writeheader()
    for row in rows:
        if key(row["text"]) not in seen:
            seen.add(key(row["text"]))
            writer.writerow(row)


if __name__ == "__main__":
    main(sys.argv[1])
